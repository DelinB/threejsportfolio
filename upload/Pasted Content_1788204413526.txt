            const LAT_STEPS = 108,

            function smoothstep(e0, e1, x) {
                let t = (x - e0) / (e1 - e0);
                t = Math.max(0, Math.min(1, t));
                return t * t * (3 - 2 * t);
            }

            function bandValue(theta, phi, bandAngle, v0, width, uFreq, uPhase, lengthLo, lengthHi) {
                const ca = Math.cos(bandAngle),
                    sa = Math.sin(bandAngle);
                const u = theta * ca + phi * sa;
                const v = -theta * sa + phi * ca;
                let closeness = Math.max(0, 1 - Math.abs(v - v0) / width);
                closeness = Math.pow(closeness, 1.5);
                const fold = Math.cos(u * uFreq + uPhase);
                const corrug = fold * 0.5 + 0.5;
                const lengthMask = smoothstep(lengthLo - 0.4, lengthLo, u) * (1 - smoothstep(lengthHi, lengthHi + 0.4, u));
                return closeness * (0.55 + 0.55 * corrug) * lengthMask;
            }

            function ribbonField(theta, phi) {
                const b1 = bandValue(theta, phi, 50 * Math.PI / 180, -0.05, 0.32, 3.6, 0.0, -1.6, 1.05);
                const b2 = bandValue(theta, phi, 18 * Math.PI / 180, 0.55, 0.28, 4.5, 1.0, -0.6, 1.4);
                return Math.max(b1, b2);
            }

                for (let j = 0; j < LON_STEPS; j++) {

                    const x0 = ring * Math.cos(phi);
                    const z0 = ring * Math.sin(phi);
                    const ribbon = ribbonField(theta, phi);
                    particles.push({ x0, y0, z0, ribbon });
                }
            }

            function mixC(c1, c2, t, out) {

                const finalScale = sca;

                    const xB = p.x0 * bulge,
                        yB = p.y0 * bulge,
                        zB = p.z0 * bulge;
                    const x1 = xB * cosY - zB * sinY;
                    const z1 = xB * sinY + zB * cosY;
                    const y1 = yB * cosX - z1 * sinX;
                    const z2 = yB * sinX + z1 * cosX;
                    if (z2 < 0.02) continue;
                    const factor = PERSPECTIVE / (PERSPECTIVE - z2);
                    const px = cx + x1 * R * factor;
                    const py = cy + y1 * R * factor;
                    const latT = Math.max(0, Math.min(1, (1 + y1) / 2));
                    const base = baseColor(latT, tmpA);
                    const strength = Math.max(0, Math.min(1, p.ribbon));
                    const s2 = strength * strength * (3 - 2 * strength);
                    const dim = mixC(DEEPBLUE, base, 0.5, tmpB);
                    const bright = mixC(base, CREAM, 0.55, tmpC);
                    const col = mixC(dim, bright, s2, tmpD);
                    const topBoost = Math.max(0, 0.32 - latT) * 0.5;
                    const dimA = 0.34 + 0.20 * factor + topBoost;
                    const brightA = Math.min(1, 0.34 + s2 * 0.85);
                    const alpha = lerp(dimA, brightA, s2);
                    const dimSize = 0.85 * factor;
                    const brightSize = (0.9 + s2 * 1.5) * factor;
                    const size = lerp(dimSize, brightSize, s2) * scale;
                    const d = drawPool[count++];
                    d.px = px;
                    d.py = py;
                    d.z = z2;
                    d.r = col[0] | 0;
                    d.g = col[1] | 0;
                    d.b = col[2] | 0;
                    d.a = alpha;
                    d.size = size;
                }

                const visible = drawPool.slice(0, count);
                visible.sort((a, b) => a.z - b.z);

                const open = () => openModal(+row.dataset.i);
                row.addEventListener('click', open);
                row.addEventListener('keydown', e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault();
                        open(); } });

            function openModal(i) {
                const p = PROJECTS[i];
                $('#mImg').src = img(p.seed, 1400, 1000);
                $('#mImg').alt = p.title + ' — project visual';
                $('#mMeta').textContent = p.meta;
                $('#mTitle').textContent = p.title;
                $('#mBrief').textContent = p.brief;
                $('#mOutcome').textContent = p.outcome;
                $('#mQuote').textContent = p.quote;
                $('#mLesson').textContent = p.lesson;
                $('#mAsk').href = 'mailto:delinbanish@gmail.com?subject=' + encodeURIComponent('Tell me about ' + p.title);
                lastFocus = document.activeElement;
                modal.classList.add('open');
                modal.setAttribute('aria-hidden', 'false');
                document.body.classList.add('lock');
                prev.classList.remove('show');
                $('#mClose').focus();
            }

            function closeModal() {
                modal.classList.remove('open');
                modal.setAttribute('aria-hidden', 'true');
                document.body.classList.remove('lock');
                if (lastFocus) lastFocus.focus();
            }
            $$('[data-close]').forEach(el => el.addEventListener('click', closeModal));
            document.addEventListener('keydown', e => {
                if (e.key !== 'Escape') return;
                if (modal.classList.contains('open')) closeModal();
                if (document.body.classList.contains('menu-open')) {
                    document.body.classList.remove('menu-open');
                    $('#menu').setAttribute('aria-hidden', 'true');
                }
            });

            // Focus trap for modal
            document.addEventListener('keydown', function(e) {
                if (modal.classList.contains('open') && e.key === 'Tab') {
                    const focusable = modal.querySelectorAll('button, a, input, [tabindex]:not([tabindex="-1"])');
                    if (!focusable.length) return;
                    const first = focusable[0];
                    const last = focusable[focusable.length - 1];
                    if (e.shiftKey && document.activeElement === first) {
                        e.preventDefault();
                        last.focus();
                    } else if (!e.shiftKey && document.activeElement === last) {
                        e.preventDefault();
                        first.focus();
                    }
                }
            });
