  1213→                let t = (x - e0) / (e1 - e0);

  1218→            function bandValue(theta, phi, bandAngle, v0, width, uFreq, uPhase, lengthLo, lengthHi) {
  1219→                const ca = Math.cos(bandAngle),
  1220→                    sa = Math.sin(bandAngle);
  1221→                const u = theta * ca + phi * sa;
  1222→                const v = -theta * sa + phi * ca;
  1223→                let closeness = Math.max(0, 1 - Math.abs(v - v0) / width);
  1224→                closeness = Math.pow(closeness, 1.5);
  1225→                const fold = Math.cos(u * uFreq + uPhase);
  1226→                const corrug = fold * 0.5 + 0.5;
  1227→                const lengthMask = smoothstep(lengthLo - 0.4, lengthLo, u) * (1 - smoothstep(lengthHi, lengthHi + 0.4, u));
  1228→                return closeness * (0.55 + 0.55 * corrug) * lengthMask;
  1229→            }
